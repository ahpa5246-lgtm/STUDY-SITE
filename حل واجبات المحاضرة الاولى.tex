\SectionHead{1}{Mesh Current Analysis}
\textbf{Question.} Find the transfer function $G(s)=V_o(s)/V_i(s)$ of the circuit shown below.
\begin{figure}[H]\centering
\begin{circuitikz}[american,scale=.91,transform shape]
\draw (0,0) to[sV,l_=$V_i$] (0,3) to[L,l=$L_1$] (3.15,3)
      to[L,l=$L_2$] (6.3,3) to[C,l=$C$] (6.3,0) -- (0,0);
\draw (3.15,3) to[R,l=$R$] (3.15,0);
\node at (1.6,1.45) {$I_1$};\node at (4.78,1.4) {$I_2$};
\draw[-{Stealth[length=2.5mm]},BlueAccent,thin] (1.25,1.25) arc[start angle=210,end angle=-90,radius=.42];
\draw[-{Stealth[length=2.5mm]},BlueAccent,thin] (4.42,1.25) arc[start angle=210,end angle=-90,radius=.42];
\node[BlueAccent,above] at (4.76,3.73) {$+\;V_o\;-$};
\draw[BlueAccent,thin] (3.92,3.64) -- (3.92,3.42);
\draw[BlueAccent,thin] (5.62,3.64) -- (5.62,3.42);
\end{circuitikz}
\caption{Two-mesh $RLC$ circuit; $V_o$ is measured across $L_2$.}
\end{figure}
\Heading{1. Laplace-domain impedances}
\Eqs{
 Z_{L_1}&=L_1s, & Z_{L_2}&=L_2s, & Z_R&=R, & Z_C&=\frac{1}{Cs},\\
 V(s)&=Z(s)I(s), & I_R&=I_1-I_2.
}
\Heading{2. Mesh 1 (KVL)}
\Eqs{
 V_i &=L_1sI_1+R(I_1-I_2)\\
     &=L_1sI_1+RI_1-RI_2\\
 \boxed{(L_1s+R)I_1-RI_2}&=V_i. \tag{1}
}
\Heading{3. Mesh 2 (KVL)}
\Eqs{
 0&=R(I_2-I_1)+L_2sI_2+\frac{I_2}{Cs}\\
  &=RI_2-RI_1+L_2sI_2+\frac{I_2}{Cs}\\
 \boxed{-RI_1+\left(R+L_2s+\frac1{Cs}\right)I_2}&=0. \tag{2}
}
\clearpage
\Heading{4. Eliminate $I_1$}
\Eqs{
 RI_1&=\left(R+L_2s+\frac{1}{Cs}\right)I_2\\
 I_1&=\frac{R+L_2s+\frac{1}{Cs}}{R}I_2\\[3pt]
 (L_1s+R)\left(\frac{R+L_2s+\frac1{Cs}}{R}\right)I_2-RI_2&=V_i\\[3pt]
 \left[\frac{(L_1s+R)(R+L_2s+\frac1{Cs})}{R}-R\right]I_2&=V_i.
}
\Heading{5. Current ratio and output voltage}
\Eqs{
 \left[\frac{(L_1s+R)(R+L_2s+\frac1{Cs})-R^2}{R}\right]I_2&=V_i\\[3pt]
 \frac{I_2}{V_i}&=\frac{R}{(L_1s+R)(R+L_2s+\frac1{Cs})-R^2}\\[4pt]
 V_o&=Z_{L_2}I_2=L_2sI_2\\
 \frac{V_o}{V_i}&=L_2s\frac{I_2}{V_i}\\[4pt]
 &=\frac{RL_2s}{(L_1s+R)(R+L_2s+\frac1{Cs})-R^2}.
}
\Heading{6. Expand and simplify}
\Eqs{
 D(s)&=(L_1s+R)\left(R+L_2s+\frac1{Cs}\right)-R^2\\
 &=L_1Rs+L_1L_2s^2+\frac{L_1}{C}+R^2+RL_2s+\frac{R}{Cs}-R^2\\
 &=L_1L_2s^2+R(L_1+L_2)s+\frac{L_1}{C}+\frac{R}{Cs}\\[3pt]
 \frac{V_o}{V_i}&=\frac{RL_2s}{L_1L_2s^2+R(L_1+L_2)s+\frac{L_1}{C}+\frac{R}{Cs}}\\[5pt]
 &=\frac{RL_2s\,(Cs)}{\left[L_1L_2s^2+R(L_1+L_2)s+\frac{L_1}{C}+\frac{R}{Cs}\right](Cs)}\\[5pt]
 &=\frac{RL_2Cs^2}{L_1L_2Cs^3+RC(L_1+L_2)s^2+L_1s+R}.
}
\Final{\frac{V_o(s)}{V_i(s)}=\frac{L_2Cs^2}{\frac{L_1L_2C}{R}s^3+(L_1+L_2)Cs^2+\frac{L_1}{R}s+1}}

\SectionHead{2}{Mesh Current Analysis}
\textbf{Question.} Find the transfer function $G(s)=E_o(s)/E_i(s)$ of the circuit shown below.
\begin{figure}[H]\centering
\begin{circuitikz}[american,scale=.87,transform shape]
\draw (0,0) to[sV,l_=$E_i$] (0,3) -- (1.2,3) -- (1.2,4.2)
    to[R,l=$R_2$] (4.6,4.2) -- (4.6,3) -- (6,3)
    to[R,l=$R_1$] (6,1.6) to[C,l=$C_1$] (6,0) -- (0,0);
\draw (1.2,3) to[C,l=$C_2$] (4.6,3);
\node at (2.4,1.5) {$I_1$}; \node[BlueAccent] at (3.95,3.64) {$I_2$};
\node[BlueAccent] at (7.2,.8) {$E_o$};
\node[BlueAccent] at (6.7,1.15) {$+$};\node[BlueAccent] at (6.7,.2) {$-$};
\draw[BlueAccent,thin] (6,1.53)--(6.53,1.53);
\draw[BlueAccent,thin] (6,0)--(6.53,0);
\end{circuitikz}
\caption{Two-mesh $RC$ circuit; $E_o$ is measured across $C_1$.}
\end{figure}
\Heading{1. Laplace-domain impedances}
\Eqs{
 Z_{R_1}&=R_1, & Z_{R_2}&=R_2, & Z_{C_1}&=\frac1{C_1s}, & Z_{C_2}&=\frac1{C_2s},\\
 V(s)&=Z(s)I(s), & I_{C_2}&=I_1-I_2.
}
\Heading{2. Mesh 1 (KVL)}
\Eqs{
 E_i&=\frac{I_1-I_2}{C_2s}+R_1I_1+\frac{I_1}{C_1s}\\
 &=\frac{I_1}{C_2s}-\frac{I_2}{C_2s}+R_1I_1+\frac{I_1}{C_1s}\\
 \boxed{\left(R_1+\frac1{C_1s}+\frac1{C_2s}\right)I_1-\frac{I_2}{C_2s}}&=E_i.\tag{1}
}
\Heading{3. Mesh 2 (KVL)}
\Eqs{
 0&=R_2I_2+\frac{I_2-I_1}{C_2s}\\
 &=R_2I_2+\frac{I_2}{C_2s}-\frac{I_1}{C_2s}\\
 \boxed{-\frac{I_1}{C_2s}+\left(R_2+\frac1{C_2s}\right)I_2}&=0.\tag{2}
}
\clearpage
\Heading{4. Eliminate $I_2$}
\Eqs{
 -I_1+(R_2C_2s+1)I_2&=0\\
 I_1&=(R_2C_2s+1)I_2\\
 I_2&=\frac{I_1}{R_2C_2s+1}.
}
\Heading{5. Substitute into Mesh 1}
\Eqs{
 E_i&=\left(R_1+\frac1{C_1s}+\frac1{C_2s}\right)I_1-\frac{I_2}{C_2s}\\[3pt]
 &=\left(R_1+\frac1{C_1s}+\frac1{C_2s}\right)I_1-\frac{I_1}{C_2s(R_2C_2s+1)}\\[3pt]
 &=\left[R_1+\frac1{C_1s}+\frac1{C_2s}-\frac1{C_2s(R_2C_2s+1)}\right]I_1.
}
\Eqs{
 \frac1{C_2s}-\frac1{C_2s(R_2C_2s+1)}
 &=\frac{R_2C_2s+1-1}{C_2s(R_2C_2s+1)}\\
 &=\frac{R_2C_2s}{C_2s(R_2C_2s+1)}\\
 &=\frac{R_2}{R_2C_2s+1}.
}
\Eqs{
 E_i&=\left[R_1+\frac1{C_1s}+\frac{R_2}{R_2C_2s+1}\right]I_1\\[3pt]
 \frac{I_1}{E_i}&=\frac{1}{R_1+\frac1{C_1s}+\frac{R_2}{R_2C_2s+1}}.
}
\clearpage\Heading{6. Output voltage and transfer function}
\Eqs{
 E_o&=Z_{C_1}I_1=\frac{I_1}{C_1s}\\
 \frac{E_o}{E_i}&=\frac1{C_1s}\frac{I_1}{E_i}\\[4pt]
 &=\frac{\frac1{C_1s}}{R_1+\frac1{C_1s}+\frac{R_2}{R_2C_2s+1}}\\[6pt]
 &=\frac{1}{R_1C_1s+1+\frac{R_2C_1s}{R_2C_2s+1}}\\[6pt]
 &=\frac{R_2C_2s+1}{(R_1C_1s+1)(R_2C_2s+1)+R_2C_1s}.
}
\Heading{7. Expand denominator}
\Eqs{
 &(R_1C_1s+1)(R_2C_2s+1)+R_2C_1s\\
 &\quad=R_1R_2C_1C_2s^2+R_1C_1s+R_2C_2s+1+R_2C_1s\\
 &\quad=R_1R_2C_1C_2s^2+(R_1C_1+R_2C_2+R_2C_1)s+1.
}
\Final{\frac{E_o(s)}{E_i(s)}=\frac{R_2C_2s+1}{R_1R_2C_1C_2s^2+(R_1C_1+R_2C_2+R_2C_1)s+1}}

\SectionHead{3}{Mesh Current Analysis}
\textbf{Question.} Find the transfer function $G(s)=I_2(s)/E_i(s)$ of the circuit shown below.
\begin{figure}[H]\centering
\begin{circuitikz}[american,scale=.91,transform shape]
\draw (0,0) to[sV,l_=$E_i$] (0,3) to[R,l=$R_1$] (3.15,3)
      to[R,l=$R_2$] (6.3,3) to[C,l=$C$] (6.3,0)--(0,0);
\draw (3.15,3) to[L,l=$L$] (3.15,0);
\node at (1.6,1.42) {$I_1$};\node at (4.75,1.42) {$I_2$};
\end{circuitikz}
\caption{Two-mesh $RLC$ circuit; the requested output is mesh current $I_2$.}
\end{figure}
\Heading{1. Laplace-domain impedances}
\Eqs{
 Z_{R_1}&=R_1, & Z_{R_2}&=R_2, & Z_L&=Ls, & Z_C&=\frac1{Cs},\\
 V(s)&=Z(s)I(s), & I_L&=I_1-I_2.
}
\Heading{2. Mesh 1 (KVL)}
\Eqs{
 E_i&=R_1I_1+Ls(I_1-I_2)\\
 &=R_1I_1+LsI_1-LsI_2\\
 \boxed{(R_1+Ls)I_1-LsI_2}&=E_i.\tag{1}
}
\Heading{3. Mesh 2 (KVL)}
\Eqs{
 0&=Ls(I_2-I_1)+R_2I_2+\frac{I_2}{Cs}\\
 &=LsI_2-LsI_1+R_2I_2+\frac{I_2}{Cs}\\
 \boxed{-LsI_1+\left(Ls+R_2+\frac1{Cs}\right)I_2}&=0.\tag{2}
}
\Heading{4. Eliminate $I_1$}
\Eqs{
 LsI_1&=\left(Ls+R_2+\frac1{Cs}\right)I_2\\
 I_1&=\frac{Ls+R_2+\frac1{Cs}}{Ls}I_2.
}
\clearpage
\Heading{5. Substitute into Mesh 1}
\Eqs{
 E_i&=(R_1+Ls)\left(\frac{Ls+R_2+\frac1{Cs}}{Ls}\right)I_2-LsI_2\\[3pt]
 &=\left[\frac{(R_1+Ls)(Ls+R_2+\frac1{Cs})}{Ls}-Ls\right]I_2\\[3pt]
 &=\left[\frac{(R_1+Ls)(Ls+R_2+\frac1{Cs})-L^2s^2}{Ls}\right]I_2.
}
\Eqs{
 \frac{I_2}{E_i}&=\frac{Ls}{(R_1+Ls)(Ls+R_2+\frac1{Cs})-L^2s^2}.
}
\Heading{6. Expand the denominator}
\Eqs{
 D(s)&=(R_1+Ls)\left(Ls+R_2+\frac1{Cs}\right)-L^2s^2\\
 &=R_1Ls+R_1R_2+\frac{R_1}{Cs}+L^2s^2+LR_2s+\frac{L}{C}-L^2s^2\\
 &=L(R_1+R_2)s+R_1R_2+\frac{R_1}{Cs}+\frac{L}{C}.
}
\Heading{7. Clear fractions and collect terms}
\Eqs{
 \frac{I_2}{E_i}&=\frac{Ls}{L(R_1+R_2)s+R_1R_2+\frac{R_1}{Cs}+\frac{L}{C}}\\[5pt]
 &=\frac{Ls(Cs)}{\left[L(R_1+R_2)s+R_1R_2+\frac{R_1}{Cs}+\frac{L}{C}\right](Cs)}\\[5pt]
 &=\frac{LCs^2}{LC(R_1+R_2)s^2+R_1R_2Cs+R_1+Ls}\\[5pt]
 &=\frac{LCs^2}{(R_1+R_2)LCs^2+(R_1R_2C+L)s+R_1}.
}
\Final{\frac{I_2(s)}{E_i(s)}=\frac{LCs^2}{(R_1+R_2)LCs^2+(R_1R_2C+L)s+R_1}}
